name: CI

on:
  pull_request:
  push:
    branches: [develop]

jobs:
  lint:
    name: Lint
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Lint
        run: npm run lint
  tests:
    name: Tests  # the required check
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Restore
        run: dotnet restore
      - name: Build and test
        run: |
          dotnet build --configuration Release
          dotnet test --no-build --configuration Release --logger "trx;LogFileName=${{ matrix.os }}.trx"
          dotnet test --no-build --configuration Release
  release:
    runs-on: ubuntu-latest
    steps:
      - run: echo releasing
